// ============================================================
// VARIABLES GLOBALES
// ============================================================
MejorSeleccion = {}
MejorCosto = 0
MejorTamanio = 0
SolucionActual = {}
CostoActual = 0
Temperatura = 0

// ============================================================
// FUNCIÓN DE COSTO CON PENALIZACIÓN
// ============================================================
let Costo(solucion) =
    union = {};
    foreach i in solucion {
        union = union | Conjuntos[i];
    };
    noCubiertos = Universo &! union;   // diferencia: elementos no cubiertos
    M = size(Conjuntos) + 1;
    size(solucion) + M * size(noCubiertos);


// ============================================================
// GENERACIÓN DE VECINOS (AÑADIR O ELIMINAR)
// ============================================================
let GenerarVecino(solucion) =
    // Decidir entre añadir o eliminar (50% cada uno)
    if (random(2) == 0) {
        // ---- AÑADIR: elegir un conjunto no presente ----
        disponibles = {};
        i = 0;
        while (i < size(Conjuntos)) {
            if (! i in solucion) {
                disponibles = disponibles | {i};
            };
            i = i + 1;
        };
        if (size(disponibles) > 0) {
            // Elegir un elemento aleatorio de disponibles
            // Como no podemos acceder por índice, recorremos hasta el elegido
            elegido = 0;
            contar = 0;
            aleatorio = random(size(disponibles));
            foreach j in disponibles {
                if (contar == aleatorio) {
                    elegido = j;
                };
                contar = contar + 1;
            };
            solucion | {elegido};
        } else {
            solucion;   // no hay disponibles, devolver igual
        };
    } else {
        // ---- ELIMINAR: quitar un conjunto presente ----
        if (size(solucion) > 0) {
            // Elegir un elemento aleatorio de solucion
            elegido = 0;
            contar = 0;
            aleatorio = random(size(solucion));
            foreach j in solucion {
                if (contar == aleatorio) {
                    elegido = j;
                };
                contar = contar + 1;
            };
            solucion &! {elegido};   // diferencia: quitar el elegido
        } else {
            // Si está vacía, forzar añadir uno aleatorio
            solucion | {random(size(Conjuntos))};
        };
    };


// ============================================================
// RECOCIDO SIMULADO (RECURSIVO)
// ============================================================
let Recocido(sol, cost, temp, mejorSol, mejorCost, iter) =
    if (temp < 0.001 || iter > 10000) {
        // Terminar: guardar la mejor solución encontrada
        MejorSeleccion = mejorSol;
        MejorCosto = mejorCost;
        MejorTamanio = size(mejorSol);   // mejorSol siempre factible por la penalización
        ;
    } else {
        // Generar vecino y calcular su costo
        vecino = GenerarVecino(sol);
        costVecino = Costo(vecino);
        delta = costVecino - cost;

        // Criterio de aceptación de Metropolis
        if (delta < 0 || random(100) / 100 < exp(-delta / temp)) {
            sol = vecino;
            cost = costVecino;
            // Actualizar la mejor solución si mejora
            if (cost < mejorCost) {
                mejorSol = sol;
                mejorCost = cost;
            };
        };

        // Enfriar y llamada recursiva
        nuevaTemp = temp * 0.95;
        Recocido(sol, cost, nuevaTemp, mejorSol, mejorCost, iter + 1);
    };


// ============================================================
// FUNCIÓN PRINCIPAL
// ============================================================
let ResolverCobertura(Universo, Conjuntos) =
    // Inicializar con TODOS los conjuntos (solución factible)
    todos = {};
    i = 0;
    while (i < size(Conjuntos)) {
        todos = todos | {i};
        i = i + 1;
    };
    SolucionActual = todos;
    CostoActual = Costo(todos);
    Temperatura = 100;          // Temperatura inicial
    MejorSeleccion = todos;
    MejorCosto = CostoActual;
    MejorTamanio = size(todos); // pero luego se actualizará

    // Ejecutar el recocido
    Recocido(SolucionActual, CostoActual, Temperatura, MejorSeleccion, MejorCosto, 0);

    // Construir la respuesta en el formato requerido (los conjuntos reales)
    RespuestaConjuntos = {};
    foreach i in MejorSeleccion {
        RespuestaConjuntos = RespuestaConjuntos | {Conjuntos[i]};
    };
    ['MejorSeleccion, MejorSeleccion, 'MejorTamanio, MejorTamanio, 'RespuestaConjuntos, RespuestaConjuntos];

    union = {};
    foreach i in solucion {
        union = union | Conjuntos[i];

    size(solucion) + M * size(noCubiertos);

let GenerarVecino(solucion) =
    if (random(2) == 0) {
        // ---- AÑADIR ----
        disponibles = {};
        i = 0;
        while (i < size(Conjuntos)) {
            if (!(i in solucion)) {
                disponibles = disponibles | {i};
            };
            i = i + 1;
        };
        if (size(disponibles) > 0) {
            // Elegir aleatorio de disponibles
            elegido = 0;
            contar = 0;
            aleatorio = random(size(disponibles));
            foreach j in disponibles {
                if (contar == aleatorio) {
                    elegido = j;
                };
                contar = contar + 1;
            };
            solucion | {elegido};
        } else {
            solucion;
        };
    } else {
        // ---- ELIMINAR ----
        if (size(solucion) > 0) {
            elegido = 0;
            contar = 0;
            aleatorio = random(size(solucion));
            foreach j in solucion {
                if (contar == aleatorio) {
                    elegido = j;
                };
                contar = contar + 1;
            };
            solucion &! {elegido};
        } else {
            // Si está vacía, añadir uno aleatorio
            solucion | {random(size(Conjuntos))};
        };
    };
